{showNewTransferModal && (
        <div className="fixed inset-0 bg-black/60 z-50 flex items-center justify-center p-3 backdrop-blur-xs">
          <div className="bg-white rounded-lg shadow-2xl border border-slate-400 w-full max-w-4xl overflow-hidden flex flex-col max-h-[92vh]">
            <div className="bg-[#143254] text-white p-3 flex items-center justify-between shrink-0">
              <div className="flex items-center gap-2 font-bold text-sm">
                <ArrowLeftRight className="w-4 h-4 text-sky-400" />
                <span>إنشاء أمر ومستند تحويل مخزني بين الفروع (New Stock Transfer)</span>
              </div>
              <button
                onClick={() => setShowNewTransferModal(false)}
                className="text-white/80 hover:text-white p-1 hover:bg-white/10 rounded cursor-pointer"
              >
                <X className="w-4 h-4" />
              </button>
            </div>

            <div className="p-4 space-y-3 overflow-y-auto flex-1 text-xs">
              {transferError && (
                <div className="p-2.5 bg-rose-50 border border-rose-300 text-rose-800 rounded font-semibold flex items-center gap-2">
                  <AlertTriangle className="w-4 h-4 text-rose-600 shrink-0" />
                  <span>{transferError}</span>
                </div>
              )}

              {/* Source & Destination Warehouses Strip */}
              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-2 bg-slate-50 p-2.5 rounded border border-slate-200">
                <div>
                  <label className="block text-[11px] font-bold text-slate-700 mb-1">تاريخ التحويل:</label>
                  <input
                    type="date"
                    value={transferDate}
                    onChange={(e) => setTransferDate(e.target.value)}
                    className="w-full p-1.5 bg-white border border-slate-300 rounded font-mono text-xs focus:outline-none focus:border-sky-500"
                  />
                </div>

                <div>
                  <label className="block text-[11px] font-bold text-slate-700 mb-1">
                    مستودع المصدر (الصرف):
                  </label>
                  <select
                    value={fromWarehouseId}
                    onChange={(e) => setFromWarehouseId(e.target.value)}
                    className="w-full p-1.5 bg-white border border-slate-300 rounded font-bold text-xs focus:outline-none focus:border-sky-500 cursor-pointer"
                  >
                    {warehouses.map((w) => (
                      <option key={w.id} value={w.id}>
                        {w.name}
                      </option>
                    ))}
                  </select>
                </div>

                <div>
                  <label className="block text-[11px] font-bold text-slate-700 mb-1">
                    مستودع الوجهة (الاستلام):
                  </label>
                  <select
                    value={toWarehouseId}
                    onChange={(e) => setToWarehouseId(e.target.value)}
                    className="w-full p-1.5 bg-white border border-slate-300 rounded font-bold text-xs focus:outline-none focus:border-sky-500 cursor-pointer"
                  >
                    {warehouses.map((w) => (
                      <option key={w.id} value={w.id}>
                        {w.name}
                      </option>
                    ))}
                  </select>
                </div>

                <div>
                  <label className="block text-[11px] font-bold text-slate-700 mb-1">رقم المرجع / الإذن:</label>
                  <input
                    type="text"
                    value={transferRef}
                    onChange={(e) => setTransferRef(e.target.value)}
                    placeholder="مثال: REQ-099"
                    className="w-full p-1.5 bg-white border border-slate-300 rounded text-xs focus:outline-none focus:border-sky-500 font-mono"
                  />
                </div>
              </div>

              {/* Logistics and Driver Data */}
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 bg-slate-50 p-2.5 rounded border border-slate-200">
                <div>
                  <label className="block text-[11px] font-bold text-slate-700 mb-1">اسم السائق / الناقل:</label>
                  <input
                    type="text"
                    value={driverName}
                    onChange={(e) => setDriverName(e.target.value)}
                    placeholder="اسم مندوب النقل أو السائق..."
                    className="w-full p-1.5 bg-white border border-slate-300 rounded text-xs focus:outline-none focus:border-sky-500"
                  />
                </div>

                <div>
                  <label className="block text-[11px] font-bold text-slate-700 mb-1">رقم لوحة وسيلة النقل:</label>
                  <input
                    type="text"
                    value={vehiclePlate}
                    onChange={(e) => setVehiclePlate(e.target.value)}
                    placeholder="مثال: نقل 12-8874"
                    className="w-full p-1.5 bg-white border border-slate-300 rounded text-xs focus:outline-none focus:border-sky-500 font-mono"
                  />
                </div>
              </div>

              {/* Add Item Line Box */}
              <div className="p-2.5 bg-sky-50/70 border border-sky-200 rounded space-y-2">
                <div className="font-bold text-sky-900 text-xs flex items-center gap-1">
                  <Plus className="w-3.5 h-3.5 text-sky-700" />
                  <span>إضافة أصناف لأمر التحويل المخزني:</span>
                </div>

                <div className="grid grid-cols-1 sm:grid-cols-12 gap-2 items-end">
                  <div className="sm:col-span-6">
                    <label className="block text-[10px] font-bold text-slate-600 mb-0.5">اختر الصنف:</label>
                    <select
                      value={selectedProductId}
                      onChange={(e) => {
                        setSelectedProductId(e.target.value);
                        const prod = products.find((p) => p.id === e.target.value);
                        if (prod) setSelectedUnit(prod.unit);
                      }}
                      className="w-full p-1.5 bg-white border border-slate-300 rounded text-xs font-semibold focus:outline-none focus:border-sky-500 cursor-pointer"
                    >
                      {products.map((p) => {
                        const stockInSource = getProductStockInWarehouse(p, fromWarehouseId);
                        return (
                          <option key={p.id} value={p.id}>
                            {p.name} (المتاح بالمصدر: {stockInSource} {p.unit})
                          </option>
                        );
                      })}
                    </select>
                  </div>

                  <div className="sm:col-span-2">
                    <label className="block text-[10px] font-bold text-slate-600 mb-0.5">الوحدة:</label>
                    <input
                      type="text"
                      readOnly
                      value={selectedUnit}
                      className="w-full p-1.5 bg-slate-100 border border-slate-300 rounded text-xs text-center font-bold text-slate-700"
                    />
                  </div>

                  <div className="sm:col-span-2">
                    <label className="block text-[10px] font-bold text-slate-600 mb-0.5">الكمية المحولة:</label>
                    <input
                      type="number"
                      min={1}
                      value={selectedQty}
                      onChange={(e) => setSelectedQty(Math.max(1, Number(e.target.value) || 1))}
                      className="w-full p-1.5 bg-white border border-slate-300 rounded text-xs text-center font-bold font-mono focus:outline-none focus:border-sky-500"
                    />
                  </div>

                  <div className="sm:col-span-2">
                    <button
                      type="button"
                      onClick={handleAddItemToTransfer}
                      className="w-full bg-sky-700 hover:bg-sky-800 text-white font-bold p-1.5 rounded text-xs flex items-center justify-center gap-1 cursor-pointer transition shadow-xs"
                    >
                      <Plus className="w-3.5 h-3.5" />
                      <span>إضافة للقائمة</span>
                    </button>
                  </div>
                </div>
              </div>

              {/* Items Grid */}
              <div className="border border-slate-200 rounded overflow-hidden">
                <table className="w-full text-right text-xs border-collapse">
                  <thead className="bg-slate-100 text-slate-700">
                    <tr>
                      <th className="p-2 border-b">كود الصنف</th>
                      <th className="p-2 border-b">اسم الصنف</th>
                      <th className="p-2 border-b text-center">الوحدة</th>
                      <th className="p-2 border-b text-center">الكمية</th>
                      <th className="p-2 border-b text-left">التكلفة الإفرادية</th>
                      <th className="p-2 border-b text-left">الإجمالي ({currency})</th>
                      <th className="p-2 border-b text-center w-12">حذف</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-100 font-mono">
                    {transferItems.length === 0 ? (
                      <tr>
                        <td colSpan={7} className="p-6 text-center text-slate-400 font-sans italic">
                          لم يتم إضافة أي أصناف إلى أمر التحويل حتى الآن
                        </td>
                      </tr>
                    ) : (
                      transferItems.map((item) => (
                        <tr key={item.id} className="hover:bg-slate-50">
                          <td className="p-2 text-slate-600">{item.productCode}</td>
                          <td className="p-2 font-sans font-bold text-slate-900">{item.productName}</td>
                          <td className="p-2 font-sans text-center text-slate-700">{item.unit}</td>
                          <td className="p-2 text-center font-bold text-indigo-700 text-sm">{item.quantity}</td>
                          <td className="p-2 text-left">{item.unitCost.toLocaleString()}</td>
                          <td className="p-2 text-left font-bold text-slate-900">
                            {item.totalCost.toLocaleString()}
                          </td>
                          <td className="p-2 text-center">
                            <button
                              type="button"
                              onClick={() => handleRemoveTransferItem(item.id)}
                              className="text-rose-600 hover:text-rose-800 p-1 rounded hover:bg-rose-50 cursor-pointer"
                            >
                              <Trash2 className="w-3.5 h-3.5" />
                            </button>
                          </td>
                        </tr>
                      ))
                    )}
                  </tbody>
                  {transferItems.length > 0 && (
                    <tfoot className="bg-slate-50 font-bold border-t border-slate-200">
                      <tr>
                        <td colSpan={3} className="p-2 text-slate-800">
                          الإجمالي الكلي:
                        </td>
                        <td className="p-2 text-center text-indigo-700 font-mono text-sm">
                          {transferItems.reduce((s, i) => s + i.quantity, 0)}
                        </td>
                        <td className="p-2"></td>
                        <td className="p-2 text-left text-emerald-800 font-mono text-sm">
                          {transferItems.reduce((s, i) => s + i.totalCost, 0).toLocaleString()} {currency}
                        </td>
                        <td></td>
                      </tr>
                    </tfoot>
                  )}
                </table>
              </div>

              {/* Notes */}
              <div>
                <label className="block text-[11px] font-bold text-slate-700 mb-1">
                  ملاحظات وتعليمات الشحنة:
                </label>
                <textarea
                  rows={2}
                  value={transferNotes}
                  onChange={(e) => setTransferNotes(e.target.value)}
                  placeholder="ملاحظات التسليم، حالة الطرود، أو سبب المناقلة..."
                  className="w-full p-2 bg-slate-50 border border-slate-300 rounded text-xs focus:bg-white focus:outline-none focus:border-sky-500"
                />
              </div>

              {/* Action Buttons */}
              <div className="flex items-center justify-between pt-2 border-t border-slate-200">
                <div className="text-[11px] text-slate-500">
                  عند الصرف المباشر يتم حسم الكميات من مستودع المصدر وإدخالها في الترانزيت
                </div>
                <div className="flex items-center gap-2">
                  <button
                    type="button"
                    onClick={() => setShowNewTransferModal(false)}
                    className="px-3.5 py-1.5 border border-slate-300 rounded text-slate-700 hover:bg-slate-100 font-bold cursor-pointer"
                  >
                    إلغاء الأمر
                  </button>

                  <button
                    type="button"
                    onClick={() => handleSubmitTransfer(false)}
                    className="px-3.5 py-1.5 bg-slate-700 hover:bg-slate-800 text-white rounded font-bold shadow-xs transition flex items-center gap-1 cursor-pointer"
                  >
                    <Clock className="w-3.5 h-3.5" />
                    <span>حفظ كمسودة [Draft]</span>
                  </button>

                  <button
                    type="button"
                    onClick={() => handleSubmitTransfer(true)}
                    className="px-4 py-1.5 bg-emerald-700 hover:bg-emerald-800 text-white rounded font-bold shadow-xs transition flex items-center gap-1 cursor-pointer"
                  >
                    <Truck className="w-3.5 h-3.5" />
                    <span>صرف وإرسال الشحنة فـوراً</span>
                  </button>
                </div>
              </div>
            </div>
          </div>
        </div>
      )}
